cmake_minimum_required(VERSION 3.16)
project(st3821671 VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(BUILD_ROS2 "Build the ROS 2 node and install launch resources" ON)
find_package(Eigen3 3.3 REQUIRED NO_MODULE)

add_library(lidar_odometry_core src/core.cpp)
target_include_directories(lidar_odometry_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
target_link_libraries(lidar_odometry_core PUBLIC Eigen3::Eigen)
target_compile_options(lidar_odometry_core PRIVATE -Wall -Wextra -Wpedantic)

if(BUILD_ROS2)
  find_package(ament_cmake REQUIRED)
  find_package(rclcpp REQUIRED)
  find_package(rcl_interfaces REQUIRED)
  find_package(sensor_msgs REQUIRED)
  find_package(geometry_msgs REQUIRED)
  find_package(nav_msgs REQUIRED)
  find_package(tf2_ros REQUIRED)
  add_executable(odom_node src/odom_node.cpp)
  target_link_libraries(odom_node PRIVATE lidar_odometry_core)
  ament_target_dependencies(odom_node rclcpp rcl_interfaces sensor_msgs geometry_msgs nav_msgs tf2_ros)
  target_compile_options(odom_node PRIVATE -Wall -Wextra -Wpedantic)
  install(TARGETS odom_node DESTINATION lib/${PROJECT_NAME})
  install(TARGETS lidar_odometry_core EXPORT lidar_odometry_targets
    ARCHIVE DESTINATION lib LIBRARY DESTINATION lib)
  install(DIRECTORY include/ DESTINATION include)
  install(DIRECTORY launch config rviz DESTINATION share/${PROJECT_NAME})
  ament_export_targets(lidar_odometry_targets HAS_LIBRARY_TARGET)
  ament_export_dependencies(Eigen3)
else()
  include(CTest)
endif()

if(BUILD_TESTING)
  enable_testing()
  add_executable(core_tests test/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE lidar_odometry_core)
  target_compile_options(core_tests PRIVATE -Wall -Wextra -Wpedantic)
  add_test(NAME core_tests COMMAND core_tests)
endif()

if(BUILD_ROS2)
  ament_package()
endif()
